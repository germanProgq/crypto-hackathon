7A. Direct WebSocket Bid Path (bypass HTTP entirely)

File: src/services/web/wsBidHandler.ts

```typescript
import { Redis } from 'ioredis'
import { WebSocket } from '@fastify/websocket'

interface WsBidRequest {
  type: 'place_bid'
  auctionId: string
  amount: number
  idempotencyKey: string
  requestId: string
}

interface WsBidResponse {
  type: 'bid_result'
  requestId: string
  success: boolean
  bidId?: string
  rank?: number
  error?: string
  latencyMs: number
}

export class TurboBidHandler {
  private bidLuaScript: string

  constructor(
    private redis: Redis,
    private connections: Map<string, WebSocket>
  ) {
    this.bidLuaScript = `
      local userId = ARGV[1]
      local auctionId = ARGV[2]
      local amount = tonumber(ARGV[3])
      local idempKey = ARGV[4]
      local ts = tonumber(ARGV[5])

      local balanceKey = 'turbo:balance:' .. userId
      local rankingKey = 'turbo:ranking:' .. auctionId
      local idempKeyFull = 'turbo:idemp:' .. idempKey

      -- Idempotency check
      if redis.call('EXISTS', idempKeyFull) == 1 then
        return {'duplicate', redis.call('GET', idempKeyFull)}
      end

      -- Balance check
      local available = tonumber(redis.call('HGET', balanceKey, 'available') or 0)
      local currentBid = tonumber(redis.call('ZSCORE', rankingKey, userId) or 0)
      local delta = amount - currentBid

      if delta <= 0 then
        return {'error', 'bid_not_higher'}
      end

      if available < delta then
        return {'error', 'insufficient_funds', available}
      end

      -- Atomic update
      redis.call('HINCRBY', balanceKey, 'available', -delta)
      redis.call('HINCRBY', balanceKey, 'held', delta)

      local score = amount * 1000000000 + (1000000000 - ts % 1000000000)
      redis.call('ZADD', rankingKey, score, userId)

      local rank = redis.call('ZREVRANK', rankingKey, userId)
      local bidId = auctionId .. ':' .. ts .. ':' .. userId

      redis.call('SETEX', idempKeyFull, 86400, bidId)

      redis.call('LPUSH', 'turbo:sync:queue', cjson.encode({
        bidId = bidId,
        auctionId = auctionId,
        userId = userId,
        amount = amount,
        ts = ts
      }))

      return {'success', bidId, rank + 1, amount}
    `
  }

  async handleBid(ws: WebSocket, userId: string, req: WsBidRequest): Promise<void> {
    const startTime = process.hrtime.bigint()

    try {
      const result = await this.redis.eval(
        this.bidLuaScript,
        0,
        userId,
        req.auctionId,
        req.amount.toString(),
        req.idempotencyKey,
        Date.now().toString()
      ) as string[]

      const latencyMs = Number(process.hrtime.bigint() - startTime) / 1_000_000

      if (result[0] === 'success') {
        ws.send(JSON.stringify({
          type: 'bid_result',
          requestId: req.requestId,
          success: true,
          bidId: result[1],
          rank: parseInt(result[2]),
          amount: parseFloat(result[3]),
          latencyMs: Math.round(latencyMs * 100) / 100
        }))

        this.broadcastBidUpdate(req.auctionId, userId, parseFloat(result[3]), parseInt(result[2]))
      } else if (result[0] === 'duplicate') {
        ws.send(JSON.stringify({
          type: 'bid_result',
          requestId: req.requestId,
          success: true,
          bidId: result[1],
          duplicate: true,
          latencyMs: Math.round(latencyMs * 100) / 100
        }))
      } else {
        ws.send(JSON.stringify({
          type: 'bid_result',
          requestId: req.requestId,
          success: false,
          error: result[1],
          latencyMs: Math.round(latencyMs * 100) / 100
        }))
      }
    } catch (err) {
      const latencyMs = Number(process.hrtime.bigint() - startTime) / 1_000_000
      ws.send(JSON.stringify({
        type: 'bid_result',
        requestId: req.requestId,
        success: false,
        error: 'internal_error',
        latencyMs: Math.round(latencyMs * 100) / 100
      }))
    }
  }

  private broadcastBidUpdate(auctionId: string, userId: string, amount: number, rank: number) {
    this.redis.publish(`auction:${auctionId}:bids`, JSON.stringify({
      type: 'bid_placed',
      auctionId,
      userId,
      amount,
      rank,
      timestamp: new Date().toISOString()
    }))
  }
}
```

7B. Connection Pooling & Memory Optimization

```typescript
// src/services/web/wsConnectionPool.ts
import { WebSocket } from '@fastify/websocket'

export class ConnectionPool {
  private connections = new Map<string, WebSocket>()
  private roomSubscriptions = new Map<string, Set<string>>()
  private userRooms = new Map<string, Set<string>>()

  private pingBuffer = Buffer.from(JSON.stringify({ type: 'ping' }))
  private pongBuffer = Buffer.from(JSON.stringify({ type: 'pong' }))

  add(userId: string, ws: WebSocket) {
    const existing = this.connections.get(userId)
    if (existing && existing.readyState === WebSocket.OPEN) {
      existing.close(1000, 'new_connection')
    }
    this.connections.set(userId, ws)
    this.userRooms.set(userId, new Set())
    ws.binaryType = 'nodebuffer'
  }

  remove(userId: string) {
    const rooms = this.userRooms.get(userId)
    if (rooms) {
      for (const room of rooms) {
        this.roomSubscriptions.get(room)?.delete(userId)
      }
    }
    this.userRooms.delete(userId)
    this.connections.delete(userId)
  }

  joinRoom(userId: string, room: string) {
    if (!this.roomSubscriptions.has(room)) {
      this.roomSubscriptions.set(room, new Set())
    }
    this.roomSubscriptions.get(room)!.add(userId)
    this.userRooms.get(userId)?.add(room)
  }

  broadcastToRoom(room: string, message: Buffer | string) {
    const subscribers = this.roomSubscriptions.get(room)
    if (!subscribers) return

    const data = typeof message === 'string' ? message : message.toString()

    for (const userId of subscribers) {
      const ws = this.connections.get(userId)
      if (ws && ws.readyState === WebSocket.OPEN) {
        ws.send(data)
      }
    }
  }

  get stats() {
    return {
      connections: this.connections.size,
      rooms: this.roomSubscriptions.size,
      avgRoomSize: this.roomSubscriptions.size > 0
        ? Array.from(this.roomSubscriptions.values()).reduce((a, b) => a + b.size, 0) / this.roomSubscriptions.size
        : 0
    }
  }
}
```

```typescript
export class CryptoAuctionWsClient {
  private ws: WebSocket | null = null
  private reconnectAttempts = 0
  private maxReconnectAttempts = 5
  private eventHandlers = new Map<string, Set<Function>>()

  constructor(private config: {
    url: string
    token: string
    onConnect?: () => void
    onDisconnect?: () => void
  }) {}

  connect(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(`${this.config.url}?token=${this.config.token}`)

      this.ws.onopen = () => {
        this.reconnectAttempts = 0
        this.config.onConnect?.()
        resolve()
      }

      this.ws.onmessage = (event) => {
        const data = JSON.parse(event.data)
        this.emit(data.type, data)
      }

      this.ws.onclose = () => {
        this.config.onDisconnect?.()
        this.attemptReconnect()
      }

      this.ws.onerror = reject
    })
  }

  placeBid(params: {
    auctionId: string
    amount: number
    idempotencyKey?: string
  }): Promise<BidResult> {
    const requestId = crypto.randomUUID()

    return new Promise((resolve, reject) => {
      const handler = (data: any) => {
        if (data.requestId === requestId) {
          this.off('bid_result', handler)
          if (data.success) resolve(data)
          else reject(new BidError(data.error))
        }
      }

      this.on('bid_result', handler)

      this.send({
        type: 'place_bid',
        requestId,
        auctionId: params.auctionId,
        amount: params.amount,
        idempotencyKey: params.idempotencyKey ?? crypto.randomUUID()
      })

      setTimeout(() => {
        this.off('bid_result', handler)
        reject(new TimeoutError('Bid request timed out'))
      }, 5000)
    })
  }

  onBidUpdate(auctionId: string, callback: (bid: BidUpdate) => void) {
    this.joinRoom(`auction:${auctionId}`)
    this.on('bid_placed', (data) => {
      if (data.auctionId === auctionId) callback(data)
    })
  }

  onOutbid(callback: (alert: OutbidAlert) => void) {
    this.on('outbid', callback)
  }

  private send(data: object) { this.ws?.send(JSON.stringify(data)) }
  private emit(event: string, data: any) {
    this.eventHandlers.get(event)?.forEach(fn => fn(data))
  }
  on(event: string, handler: Function) {
    if (!this.eventHandlers.has(event)) this.eventHandlers.set(event, new Set())
    this.eventHandlers.get(event)!.add(handler)
  }
  off(event: string, handler: Function) {
    this.eventHandlers.get(event)?.delete(handler)
  }
  private joinRoom(room: string) { this.send({ type: 'join_room', room }) }
  private attemptReconnect() {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++
      setTimeout(() => this.connect(), 1000 * this.reconnectAttempts)
    }
  }
}
```

10B. Webhook Notifications

File: src/services/workers/webhookDispatcher.ts

```typescript
interface WebhookConfig {
  url: string
  secret: string
  events: ('bid_placed' | 'round_complete' | 'withdrawal_complete')[]
}

export class WebhookDispatcher {
  async dispatch(event: string, payload: object) {
    const subscribers = await this.getSubscribers(event)

    for (const config of subscribers) {
      const signature = this.sign(payload, config.secret)

      await fetch(config.url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Webhook-Signature': signature,
          'X-Event-Type': event
        },
        body: JSON.stringify(payload)
      }).catch(err => {
        logger.warn({ err, url: config.url }, 'Webhook delivery failed')
        this.queueRetry(config, event, payload)
      })
    }
  }

  private sign(payload: object, secret: string): string {
    return crypto.createHmac('sha256', secret).update(JSON.stringify(payload)).digest('hex')
  }
}
```

```typescript
interface FeatureFlags {
  fastBidPath: boolean
  mlAnomalyDetection: boolean
  cutoffPricing: boolean
  webhookNotifications: boolean
  graphqlApi: boolean
}

export async function getFeatureFlags(userId?: string): Promise<FeatureFlags> {
  const defaults: FeatureFlags = {
    fastBidPath: true,
    mlAnomalyDetection: true,
    cutoffPricing: true,
    webhookNotifications: false,
    graphqlApi: false
  }

  if (userId) {
    const userFlags = await redis.hgetall(`flags:user:${userId}`)
    return { ...defaults, ...userFlags }
  }

  const globalFlags = await redis.hgetall('flags:global')
  return { ...defaults, ...globalFlags }
}
```
